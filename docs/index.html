@flowchart-sequence-designer/angular

Embed a powerful SVG flowchart & sequence-diagram editor in any Angular 16+ application with standalone components.

$ npm install @flowchart-sequence-designer/angular flowchart-sequence-designer react react-dom

Components

FsdDiagramComponent

<fsd-diagram>

Full diagram editor supporting flowchart, question-tree, and journey variants.

FsdSequenceComponent

<fsd-sequence>

Sequence diagram editor with drag-to-reorder messages and inline editing.

FsdToolbarComponent

<fsd-toolbar>

Standalone export/import toolbar for custom editor layouts.

FsdStepEditorComponent

<fsd-step-editor>

Node property editor panel for detailed node configuration.

Usage Examples

import { Component } from '@angular/core';
import { FsdDiagramComponent } from '@flowchart-sequence-designer/angular';
import type { DiagramModel } from '@flowchart-sequence-designer/angular';

@Component({
  selector: 'app-flowchart',
  standalone: true,
  imports: [FsdDiagramComponent],
  template: \`
    <fsd-diagram
      [height]="600"
      [theme]="'dark'"
      [variant]="'flowchart'"
      [allowImport]="true"
      (modelChange)="onModelChange($event)"
      (exportEvent)="onExport($event)"
    />
  \`,
})
export class FlowchartComponent {
  onModelChange(model: DiagramModel) {
    console.log('Nodes:', model.nodes?.length);
  }

  onExport(e: { format: string; content: string | Blob }) {
    // Handle SVG, PNG, Mermaid, PlantUML, JSON exports
  }
}
import { Component } from '@angular/core';
import {
  FsdSequenceComponent,
  presetSequenceModel,
} from '@flowchart-sequence-designer/angular';

@Component({
  selector: 'app-sequence',
  standalone: true,
  imports: [FsdSequenceComponent],
  template: \`
    <fsd-sequence
      [initialModel]="model"
      [height]="500"
      [theme]="'auto'"
      (modelChange)="model = $event"
    />
  \`,
})
export class SequenceComponent {
  model = presetSequenceModel();
}
import { Component } from '@angular/core';
import {
  FsdDiagramComponent,
  FsdToolbarComponent,
  FsdStepEditorComponent,
} from '@flowchart-sequence-designer/angular';

@Component({
  selector: 'app-custom-layout',
  standalone: true,
  imports: [FsdDiagramComponent, FsdToolbarComponent, FsdStepEditorComponent],
  template: \`
    <div class="editor-layout">

        [allowedExports]="['svg', 'png', 'mermaid']"
        [allowImport]="true"

      />
      <div class="editor-body">
        <fsd-diagram

          [height]="400"

        />
        <fsd-step-editor
          *ngIf="selectedNodeId"
          [nodeId]="selectedNodeId"
          [model]="model"

        />
      </div>
    </div>
  \`,
})
export class CustomLayoutComponent {
  model: any;
  selectedNodeId: string | null = null;

  handleExport(format: string) { /* ... */ }
  handleImport(text: string) { /* ... */ }
}

Architecture

┌─────────────────────────────────────────────────────────┐
│ Angular Application (Zone.js) │
│ ┌───────────────────────────────────────────────────┐ │
│ │ <fsd-diagram> (OnPush, standalone) │ │
│ │ ┌─────────────────────────────────────────────┐ │ │
│ │ │ React Bridge (runs outside NgZone) │ │ │
│ │ │ ┌───────────────────────────────────────┐ │ │ │
│ │ │ │ React Root → DiagramEditor │ │ │ │
│ │ │ │ (SVG canvas, toolbar, import/export) │ │ │ │
│ │ │ └───────────────────────────────────────┘ │ │ │
│ │ └─────────────────────────────────────────────┘ │ │
│ │ onChange → NgZone.run() → modelChange.emit() │ │
│ └───────────────────────────────────────────────────┘ │
└─────────────────────────────────────────────────────────┘

Key Design Decisions

  • OnPush compatible — all components use ChangeDetectionStrategy.OnPush
  • Zone-optimized — React renders run outside NgZone; only outputs trigger CD
  • Lazy-loaded — React editor loaded via dynamic import() (tree-shake friendly)
  • Error handled — Loading and error states shown while editor initializes
  • Model reset — Changing initialModel after init re-mounts the full editor
  • Minimal overhead — Bridge is ~1KB; React+ReactDOM add ~45KB gzipped
